const ALLOWED_ORIGIN = "https://kazereich.github.io"; const CACHE_TTL = 600; // 10分 function corsHeaders(origin) { return { "Access-Control-Allow-Origin": ALLOWED_ORIGIN, "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Vary": "Origin" }; } function jsonResponse(data, status, origin) { return new Response(JSON.stringify(data), { status, headers: { ...corsHeaders(origin), "Content-Type": "application/json; charset=UTF-8", "Cache-Control": "no-store" } }); } export default { async fetch(request, env, ctx) { const origin = request.headers.get("Origin") || ""; if (origin && origin !== ALLOWED_ORIGIN) { return new Response("Forbidden", { status: 403 }); } if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: { ...corsHeaders(origin), "Access-Control-Max-Age": "86400" } }); } if (request.method !== "GET") { return jsonResponse({ error: "Method not allowed" }, 405, origin); } const url = new URL(request.url); const match = url.pathname.match( /^\/api\/minecraft\/([0-9a-f-]{32,36})$/i ); if (!match) { return jsonResponse({ error: "Invalid URL" }, 400, origin); } const uuid = match[1].replace(/-/g, "").toLowerCase(); if (!/^[0-9a-f]{32}$/.test(uuid)) { return jsonResponse({ error: "Invalid UUID" }, 400, origin); } const cacheKey = new Request( `https://minecraft-uuid-cache.example/${uuid}` ); const cache = caches.default; // キャッシュ確認 const cached = await cache.match(cacheKey); if (cached) { const headers = new Headers(cached.headers); Object.entries(corsHeaders(origin)).forEach(([key, value]) => { headers.set(key, value); }); headers.set("X-Cache", "HIT"); return new Response(cached.body, { status: cached.status, headers }); } try { const apiResponse = await fetch( `https://api.ashcon.app/mojang/v2/user/${uuid}`, { headers: { "Accept": "application/json" } } ); if (!apiResponse.ok) { if (apiResponse.status === 404) { return jsonResponse( { error: "Player not found" }, 404, origin ); } if (apiResponse.status === 429) { return jsonResponse( { error: "Upstream rate limit" }, 429, origin ); } return jsonResponse( { error: "Upstream API error", status: apiResponse.status }, 502, origin ); } const data = await apiResponse.json(); if (!data.username) { return jsonResponse( { error: "Username not found" }, 502, origin ); } const result = { username: data.username, uuid: uuid }; const response = new Response(JSON.stringify(result), { status: 200, headers: { "Content-Type": "application/json; charset=UTF-8", "Cache-Control": `public, max-age=${CACHE_TTL}` } }); // 成功した結果だけを10分間キャッシュ ctx.waitUntil(cache.put(cacheKey, response.clone())); const headers = new Headers(response.headers); Object.entries(corsHeaders(origin)).forEach(([key, value]) => { headers.set(key, value); }); headers.set("X-Cache", "MISS"); return new Response(response.body, { status: 200, headers }); } catch (error) { return jsonResponse( { error: "API connection failed" }, 502, origin ); } } };
